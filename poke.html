import { useState, useEffect } from "react"; export default function App() { const [cards, setCards] = useState([]); const [progress, setProgress] = useState(0); useEffect(() => { const stored = localStorage.getItem("cards"); if (stored) { const parsed = JSON.parse(stored); setCards(parsed); updateProgress(parsed); } }, []); const updateProgress = (data) => { const total = data.length; const owned = data.filter((c) => parseInt(c.quantita || 0) > 0).length; setProgress(Math.round((owned / total) * 100)); }; const handleFile = (e) => { const file = e.target.files[0]; const reader = new FileReader(); reader.onload = (event) => { const text = event.target.result; const rows = text.split("\n").slice(1); // skip header const parsed = rows.map((row) => { const cols = row.split(","); return { numero: cols[3], prezzo: cols[4]?.trim(), quantita: cols[5]?.trim(), posseduta: cols[0]?.toLowerCase().includes("true"), }; }); setCards(parsed); localStorage.setItem("cards", JSON.stringify(parsed)); updateProgress(parsed); }; reader.readAsText(file); }; const updateQuantity = (index, delta) => { const newCards = [...cards]; const current = parseInt(newCards[index].quantita || 0); newCards[index].quantita = current + delta; setCards(newCards); localStorage.setItem("cards", JSON.stringify(newCards)); updateProgress(newCards); }; return (

Inventario Carte

Completamento: {progress}%

{cards.map((card, index) => (
#{card.numero}
Prezzo: {card.prezzo}
Quantità: {card.quantita}
))}
); }
